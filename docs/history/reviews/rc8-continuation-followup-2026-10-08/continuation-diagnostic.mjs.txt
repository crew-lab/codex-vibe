async function connect(backend,label){
 home=`${root}/${label}`;await mkdir(home,{recursive:true,mode:0o700});
 await writeFile(`${home}/config.toml`,(await readFile(`${root}/home/config.toml`,'utf8')).replace('backend = "programmatic"',`backend = "${backend}"`),{mode:0o600});
 client=new Client({name:'rc8-phase-d',version:'1.0.0'});

 transport.stderr?.on('data',()=>{});
 await client.connect(transport);
 await record(`tools-${backend}`,{tools:(await client.listTools()).tools.map(t=>t.name),server_pid:transport.pid,home});
}
async function raw(name,args){const t=performance.now();const reply=await client.callTool({name,arguments:args},{timeout:600000});const text=reply.content.find(c=>c.type==='text')?.text??'{}';let data=reply.structuredContent;if(!data){try{data=JSON.parse(text);}catch{data={error:{code:text.match(/VSUP_[A-Z_]+/)?.[0],message:text}};}}return {data,isError:!!reply.isError,ms:Math.round(performance.now()-t)};}
async function call(name,args){const r=await raw(name,args);if(r.isError)throw Error(`${name}: ${JSON.stringify(r.data)}`);return r.data;}
async function settle(id){let after=0;for(let n=0;n<100;n++){const s=await call('vibe_status',{run_id:id,after_seq:after,wait_seconds:30,max_events:100});after=s.next_after_seq??after;if(s.pending_request)throw Error('Unexpected permission/input request');if(['completed','failed','cancelled','closed'].includes(s.state)){await record('settled',{run_id:id,reply:s});if(s.state!=='completed')throw Error(`Unexpected state ${s.state}: ${JSON.stringify(s.result??s)}`);return s;} }throw Error('No settled result');}
async function start(kind,task,max_turns=20,timeout_seconds=180){const s=await call(`vibe_${kind}_start`,{cwd:workspace,task,max_turns,timeout_seconds,wait_seconds:30});active.add(s.run_id);await record('started',{run_id:s.run_id,kind,reply:s});return s.run_id;}
async function close(id,cleanup=false){const s=await call('vibe_close',{run_id:id,cleanup_worktree:cleanup});await record('closed',{run_id:id,reply:s});if(s.error)throw Error(`Close failed ${JSON.stringify(s.error)}`);if(cleanup&&!s.worktree_removed)throw Error('Worktree not removed');active.delete(id);}
async function disconnect(label){const pid=transport.pid,t=performance.now();let timer;try{await Promise.race([client.close(),new Promise((_,rej)=>{timer=setTimeout(()=>rej(Error('Client close timeout')),10000);})]);}finally{clearTimeout(timer);}for(let n=0;n<100;n++){try{process.kill(pid,0);}catch{await record(label,{server_pid:pid,exit_ms:Math.round(performance.now()-t),handles:process._getActiveHandles().map(x=>x.constructor.name)});return;}await sleep(50);}throw Error('Server survives client close');}

Read ${path}/start.txt with read_file and follow each Next file pointer with a separate read_file call learned from the preceding result, exactly one file per model response.  if(restart){await disconnect(`${label}-before-restart`);await connect('acp','continuation-diagnostic');}
 await call('vibe_continue',{run_id:id,max_turns:10,message});const final=await settle(id);
 const full=await call('vibe_result',{run_id:id,detail:'full'});

 await record(label,{run_id:id,restart,stop_reason:final.result.stop_reason,files,result:full});
 await close(id,true);return {stop_reason:final.result.stop_reason,files};
}
